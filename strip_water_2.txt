#!/bin/bash
#SBATCH --job-name=or51_strip_solvent
#SBATCH --partition=cpu
#SBATCH --nodes=1
#SBATCH --ntasks=1
#SBATCH --cpus-per-task=1
#SBATCH --mem=4G
#SBATCH --time=04:00:00
#SBATCH --output=slurm_logs/%x-%j.out
#SBATCH --error=slurm_logs/%x-%j.err

# Put both scripts in ~/OR51_gate. Submit from there:
#   mkdir -p slurm_logs
#   sbatch strip_water.slurm
# The log directory must exist BEFORE sbatch starts the job.
# Optional: sbatch strip_water.slurm --dry-run
# To replace previous outputs: sbatch strip_water.slurm --overwrite

set -euo pipefail

PROJECT_DIR="${OR51_GATE_DIR:-$HOME/OR51_gate}"
OR51_VENV_DIR="${OR51_GATE_VENV:-$HOME/venvs/or_gate}"
OR51_PYTHON="$OR51_VENV_DIR/bin/python"
cd "$PROJECT_DIR"

# Call the environment's Python directly; no module or activation is required.
# The CPU node must still provide a compatible base Python for this environment.
unset PYTHONHOME PYTHONPATH
export OMP_NUM_THREADS=1
export OPENBLAS_NUM_THREADS=1
export MKL_NUM_THREADS=1
export PYTHONUNBUFFERED=1

echo "Job: ${SLURM_JOB_ID:-local} | Host: $(hostname)"
echo "Project: $PROJECT_DIR"
echo "Environment Python: $OR51_PYTHON"
date

# Check this node before touching output files. A venv created on another node
# may point to a different /usr/bin/python3 here.
if ! "$OR51_PYTHON" -u - <<'PY'
import sys
from pathlib import Path

print("Python:", sys.version.split()[0])
print("Executable:", sys.executable)
print("Environment:", sys.prefix)
print("Base Python:", sys.base_prefix)
if sys.prefix == sys.base_prefix:
    raise RuntimeError("The selected interpreter is not running in a virtual environment.")
config = Path(sys.prefix) / "pyvenv.cfg"
settings = dict(line.split("=", 1) for line in config.read_text().splitlines() if "=" in line)
settings = {key.strip(): value.strip() for key, value in settings.items()}
created_version = settings.get("version", "")
if created_version and tuple(map(int, created_version.split(".")[:2])) != sys.version_info[:2]:
    raise RuntimeError(f"Environment was created with Python {created_version}, "
                       f"but this node runs {sys.version.split()[0]}.")
import numpy
import mdtraj
from mdtraj.formats import XTCTrajectoryFile
print("NumPy:", numpy.__version__, "| MDTraj:", mdtraj.__version__)
print("Environment check passed.")
PY
then
    echo "ERROR: or_gate cannot run correctly on this CPU node. See the error above." >&2
    echo "Environment configuration and interpreter links:"
    cat "$OR51_VENV_DIR/pyvenv.cfg" 2>/dev/null || true
    ls -l "$OR51_VENV_DIR/bin/python" "$OR51_VENV_DIR/bin/python3" 2>/dev/null || true
    exit 1
fi

"$OR51_PYTHON" -u "$PROJECT_DIR/strip_water.py" \
    --input-root "$PROJECT_DIR/Shared_Trajs" \
    --output-root "$PROJECT_DIR/Stripped_Trajs" \
    --chunk 100 \
    "$@"

date
echo "Water and ion stripping finished."
