import os
import shlex
import shutil
from pathlib import Path


# ============================================================
# RESOLVE EXACTLY ONE FILE
# ============================================================

def resolve_single_file(folder, filename_or_pattern):
    """
    Accept either:

        converted.mae

    or a wildcard:

        *.mae

    There must be exactly ONE matching file.

    Backup files ending in .bak are ignored.
    """

    folder = Path(folder)

    # --------------------------------------------------------
    # Wildcard pattern
    # --------------------------------------------------------

    if any(char in filename_or_pattern for char in "*?[]"):

        matches = sorted(
            folder.glob(filename_or_pattern)
        )

        # Ignore backups
        matches = [
            f for f in matches
            if not f.name.endswith(".bak")
        ]

        if len(matches) == 0:

            raise FileNotFoundError(
                f"\nNo file matched:\n"
                f"  Folder  = {folder}\n"
                f"  Pattern = {filename_or_pattern}"
            )

        if len(matches) > 1:

            match_text = "\n".join(
                f"  {f.name}"
                for f in matches
            )

            raise ValueError(
                f"\nMultiple files matched "
                f"'{filename_or_pattern}':\n"
                f"{match_text}\n\n"
                f"There must be exactly one."
            )

        return matches[0]

    # --------------------------------------------------------
    # Exact filename
    # --------------------------------------------------------

    file_path = folder / filename_or_pattern

    if not file_path.exists():

        raise FileNotFoundError(
            f"\nFile does not exist:\n"
            f"{file_path}"
        )

    return file_path


# ============================================================
# PARSE POSSIBLE MAE ATOM LINE
# ============================================================

def parse_mae_atom_line(line):
    """
    Parse one possible atom record from a Maestro .mae file.

    Example beginning of an atom line:

        2323 86 -9.544124 1.715803 -6.249318 ...

    Important:
        fields[0] = atom ID

    Quoted fields such as:

        "CU "
        "CU"

    are handled correctly by shlex.split().

    Returns:

        {
            "atomid": 2323,
            "fields": [...]
        }

    or None if the line does not look like an atom record.
    """

    try:

        fields = shlex.split(
            line,
            posix=True
        )

    except ValueError:

        return None

    # Need enough fields to reasonably be an atom record
    if len(fields) < 5:

        return None

    # --------------------------------------------------------
    # First field must be atom ID
    # --------------------------------------------------------

    try:

        atomid = int(fields[0])

    except (ValueError, IndexError):

        return None

    # --------------------------------------------------------
    # Coordinates should be fields 2, 3, 4
    #
    # This helps exclude unrelated numeric lines.
    # --------------------------------------------------------

    try:

        float(fields[2])
        float(fields[3])
        float(fields[4])

    except (ValueError, IndexError):

        return None

    return {
        "atomid": atomid,
        "fields": fields,
    }


# ============================================================
# FIND AND VERIFY METAL ATOM
# ============================================================

def find_metal_atom_id_in_mae(
    mae_file,
    metal="CU"
):
    """
    Find exactly ONE verified metal atom.

    For Cu, a valid record must satisfy ALL of these:

    1. First field is an integer atom ID.

       Example:
           2323 ...

    2. At least one parsed field is exactly:

           CU

       Quoted MAE values such as "CU " become CU after
       parsing and stripping whitespace.

    3. Another parsed field must equal:

           CU2323

       where 2323 comes from the first-column atom ID.

       This cross-checks the atom number against the
       Maestro atom label.

    Example valid record:

        2323 ... "CU " "CU " ... Cu2323 ...

    If a CU-containing line has atom ID 2323 but contains
    Cu2332 rather than Cu2323, it is NOT accepted.

    There must be exactly ONE verified record.
    """

    mae_file = Path(mae_file)

    target_metal = metal.strip().upper()

    verified_matches = []

    # Lines that contain CU but fail the Cu<atomid> test
    failed_verification = []

    with open(
        mae_file,
        "r",
        encoding="utf-8",
        errors="ignore"
    ) as f:

        for line_number, line in enumerate(
            f,
            start=1
        ):

            parsed = parse_mae_atom_line(
                line
            )

            if parsed is None:

                continue

            atomid = parsed["atomid"]

            fields = parsed["fields"]

            # ------------------------------------------------
            # Normalize every individual MAE field
            #
            # Examples:
            #
            # "CU "
            # "CU"
            # Cu2323
            #
            # become:
            #
            # CU
            # CU
            # CU2323
            # ------------------------------------------------

            normalized_fields = [
                str(field).strip().upper()
                for field in fields
            ]

            # =================================================
            # CHECK 1:
            # Must contain an exact CU field
            # =================================================

            if target_metal not in normalized_fields:

                continue

            # =================================================
            # CHECK 2:
            # Cross-check atom ID against Cu<atomid>
            #
            # atom ID 2323:
            #
            # expected label = CU2323
            # =================================================

            expected_label = (
                f"{target_metal}{atomid}"
            )

            if expected_label not in normalized_fields:

                # ------------------------------------------------
                # Record other CU-like labels for debugging
                # ------------------------------------------------

                metal_labels_found = [
                    field
                    for field in normalized_fields
                    if (
                        field.startswith(target_metal)
                        and field != target_metal
                    )
                ]

                failed_verification.append(
                    {
                        "atomid": atomid,
                        "line_number": line_number,
                        "expected_label": expected_label,
                        "metal_labels_found": metal_labels_found,
                        "line": line.rstrip(),
                    }
                )

                continue

            # =================================================
            # VERIFIED MATCH
            # =================================================

            verified_matches.append(
                {
                    "atomid": atomid,
                    "line_number": line_number,
                    "atom_label": expected_label,
                    "line": line.rstrip(),
                }
            )

    # ========================================================
    # EXACTLY ONE VERIFIED MATCH
    # ========================================================

    if len(verified_matches) == 1:

        match = verified_matches[0]

        return (
            match["atomid"],
            match["line_number"],
            match["line"]
        )

    # ========================================================
    # ZERO VERIFIED MATCHES
    # ========================================================

    if len(verified_matches) == 0:

        print()
        print("=" * 70)
        print("ERROR: NO VERIFIED METAL ATOM FOUND")
        print("=" * 70)

        print()
        print(
            f"Metal searched : {target_metal}"
        )

        print(
            f"MAE file       : {mae_file}"
        )

        # ----------------------------------------------------
        # CU records existed but label verification failed
        # ----------------------------------------------------

        if failed_verification:

            print()
            print(
                f"Found {len(failed_verification)} "
                f"{target_metal}-containing atom record(s),"
            )

            print(
                "but the atom-ID cross-check failed."
            )

            for candidate in failed_verification:

                print()
                print("-" * 70)

                print(
                    f"MAE line       : "
                    f"{candidate['line_number']}"
                )

                print(
                    f"First-field ID : "
                    f"{candidate['atomid']}"
                )

                print(
                    f"Expected label : "
                    f"{candidate['expected_label']}"
                )

                if candidate["metal_labels_found"]:

                    print(
                        "Metal-like labels actually found:"
                    )

                    for label in candidate[
                        "metal_labels_found"
                    ]:

                        print(
                            f"  {label}"
                        )

                else:

                    print(
                        "No Cu<atomid>-style label was found."
                    )

                print()
                print("Full MAE line:")

                print(
                    candidate["line"]
                )

        else:

            print()
            print(
                f"No atom record containing an exact "
                f"'{target_metal}' field was found."
            )

        print()
        print("=" * 70)

        raise ValueError(
            f"\nCould not verify a unique "
            f"{target_metal} atom."
        )

    # ========================================================
    # MULTIPLE VERIFIED MATCHES
    #
    # Never choose the first automatically.
    # ========================================================

    print()
    print("=" * 70)
    print("ERROR: MULTIPLE VERIFIED METAL ATOMS FOUND")
    print("=" * 70)

    print()
    print(
        f"Found {len(verified_matches)} verified "
        f"{target_metal} atoms."
    )

    for match in verified_matches:

        print()
        print("-" * 70)

        print(
            f"MAE line   : "
            f"{match['line_number']}"
        )

        print(
            f"Atom ID    : "
            f"{match['atomid']}"
        )

        print(
            f"Atom label : "
            f"{match['atom_label']}"
        )

        print()
        print("Full MAE line:")

        print(
            match["line"]
        )

    print()
    print("=" * 70)

    raise ValueError(
        f"\nFound multiple verified {target_metal} atoms.\n"
        "The script will NOT choose one automatically."
    )


# ============================================================
# MODIFY METAL_CONSTRAINTS
# ============================================================

def modify_metal_constraint(
    input_file,
    atomid,
    metal_label="metal1",
    make_backup=True
):
    """
    Replace exactly ONE METAL_CONSTRAINTS line.

    Example:

        METAL_CONSTRAINTS   "metal1 2323",

    There must be exactly one METAL_CONSTRAINTS line.
    """

    input_file = Path(
        input_file
    )

    if not input_file.exists():

        raise FileNotFoundError(
            f"\nConstraint file not found:\n"
            f"{input_file}"
        )

    # --------------------------------------------------------
    # Read input file
    # --------------------------------------------------------

    with open(
        input_file,
        "r",
        encoding="utf-8"
    ) as f:

        lines = f.readlines()

    # --------------------------------------------------------
    # Find all METAL_CONSTRAINTS lines
    # --------------------------------------------------------

    indices = []

    for i, line in enumerate(lines):

        if line.lstrip().startswith(
            "METAL_CONSTRAINTS"
        ):

            indices.append(i)

    # ========================================================
    # ZERO CONSTRAINT LINES
    # ========================================================

    if len(indices) == 0:

        raise ValueError(
            f"\nCould not find METAL_CONSTRAINTS in:\n"
            f"{input_file}"
        )

    # ========================================================
    # MULTIPLE CONSTRAINT LINES
    # ========================================================

    if len(indices) > 1:

        line_numbers = [
            i + 1
            for i in indices
        ]

        raise ValueError(
            f"\nFound {len(indices)} "
            f"METAL_CONSTRAINTS lines in:\n"
            f"{input_file}\n"
            f"Line numbers: {line_numbers}\n\n"
            f"There should be exactly one."
        )

    # ========================================================
    # BACKUP
    # ========================================================

    if make_backup:

        backup_file = Path(
            str(input_file) + ".bak"
        )

        shutil.copy2(
            input_file,
            backup_file
        )

        print()
        print("Backup created:")
        print(backup_file)

    # ========================================================
    # BUILD NEW CONSTRAINT
    # ========================================================

    new_constraint = (
        f"{metal_label} {atomid}"
    )

    i = indices[0]

    original_line = lines[i]

    # Preserve original indentation
    indentation = original_line[
        :len(original_line)
        - len(original_line.lstrip())
    ]

    lines[i] = (
        f'{indentation}'
        f'METAL_CONSTRAINTS   '
        f'"{new_constraint}",\n'
    )

    # ========================================================
    # WRITE UPDATED FILE
    # ========================================================

    with open(
        input_file,
        "w",
        encoding="utf-8"
    ) as f:

        f.writelines(lines)

    return new_constraint


# ============================================================
# MAIN WORKFLOW
# ============================================================

def setup_metal_constraint(
    folder,
    mae_filename,
    constraint_filename,
    metal="CU",
    metal_label="metal1",
    make_backup=True
):
    """
    Complete workflow:

    1. Resolve exactly one MAE file.
    2. Find an exact metal field such as CU.
    3. Read the first-column atom ID.
    4. Cross-check that Cu<atomid> exists on the same record.
    5. Require exactly one verified atom.
    6. Resolve exactly one Glide input file.
    7. Update METAL_CONSTRAINTS.
    """

    folder = Path(folder)

    print()
    print("=" * 70)
    print("GRID METAL CONSTRAINT SETUP")
    print("=" * 70)

    print(
        f"Folder           : {folder}"
    )

    print(
        f"Metal            : {metal}"
    )

    print(
        f"Constraint label : {metal_label}"
    )

    # ========================================================
    # 1. MAE FILE
    # ========================================================

    mae_file = resolve_single_file(
        folder,
        mae_filename
    )

    print(
        f"MAE file         : {mae_file.name}"
    )

    # ========================================================
    # 2. FIND + VERIFY METAL ATOM
    # ========================================================

    (
        atomid,
        line_number,
        mae_line
    ) = find_metal_atom_id_in_mae(
        mae_file=mae_file,
        metal=metal
    )

    expected_label = (
        f"{metal.upper()}{atomid}"
    )

    print()
    print("Metal atom verified:")

    print(
        f"Atom ID          : {atomid}"
    )

    print(
        f"Expected label    : {expected_label}"
    )

    print(
        f"MAE line number  : {line_number}"
    )

    print()
    print("Matched MAE line:")

    print(
        mae_line
    )

    # ========================================================
    # 3. CONSTRAINT INPUT FILE
    # ========================================================

    constraint_file = resolve_single_file(
        folder,
        constraint_filename
    )

    print()
    print(
        f"Constraint file  : "
        f"{constraint_file.name}"
    )

    # ========================================================
    # 4. MODIFY METAL CONSTRAINT
    # ========================================================

    new_constraint = modify_metal_constraint(
        input_file=constraint_file,
        atomid=atomid,
        metal_label=metal_label,
        make_backup=make_backup
    )

    # ========================================================
    # FINAL REPORT
    # ========================================================

    print()
    print("=" * 70)
    print("METAL CONSTRAINT UPDATED")
    print("=" * 70)

    print(
        f'METAL_CONSTRAINTS   '
        f'"{new_constraint}",'
    )

    print("=" * 70)
    print()

    # ========================================================
    # RETURN RESULTS
    # ========================================================

    return {
        "metal": metal,
        "atomid": atomid,
        "atom_label": expected_label,
        "mae_file": str(mae_file),
        "mae_line_number": line_number,
        "constraint_file": str(constraint_file),
        "constraint": new_constraint,
    }
