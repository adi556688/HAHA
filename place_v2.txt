#!/usr/bin/env python3
"""
Place a Cu(I) atom relative to two cysteine sulfur atoms selected through
GPCR generic-position mappings, and write a PDB coordinate tail to a text file.

The first non-comment line of the output is only PDB columns 31-78:
    x, y, z, occupancy, B-factor, and element
so the HETATM serial/name/residue/chain/residue-number fields can be edited later.

Algorithm
---------
For fixed sulfur coordinates S1 and S2, the target Cu-S distances define two
spheres. If the spheres intersect, their intersection is a circle (not a unique
point). The script samples that circle and chooses the point in the least crowded
local protein environment. If the spheres do not intersect, it reports the
least-squares point minimizing

    sqrt((|Cu-S1| - r1)^2 + (|Cu-S2| - r2)^2)

and writes a warning that the fixed cysteine conformation cannot reproduce the
reference geometry.

Version: 3.1

Usage:
    python3 place_cu_site1_from_generic_positions_cli.py OR2M3 OR2M3_Prep.pdb

Optional output filename:
    python3 place_cu_site1_from_generic_positions_cli.py \
        OR2M3 OR2M3_Prep.pdb \
        -o OR2M3_Cu.txt

Dependencies:
    numpy
    pandas
    openpyxl
"""

@dataclass(frozen=True)
class ResidueInfo:
    generic_position: str
    residue_label: str
    residue_name_1: str
    residue_number: int
    segment: str


@dataclass(frozen=True)
class AtomRecord:
    record_name: str
    serial: int
    atom_name: str
    altloc: str
    residue_name: str
    chain_id: str
    residue_number: int
    insertion_code: str
    coord: np.ndarray
    occupancy: float
    b_factor: float
    element: str

    @property
    def identifier(self) -> str:
        ins = self.insertion_code.strip()

        if ins:
            resid = f"{self.residue_number}{ins}"
        else:
            resid = str(self.residue_number)

        if self.altloc:
            alt = f" altloc={self.altloc}"
        else:
            alt = ""

        return (
            f"{self.record_name} "
            f"serial={self.serial} "
            f"{self.atom_name} "
            f"{self.residue_name} "
            f"chain={self.chain_id!r} "
            f"resid={resid}"
            f"{alt}"
        )


@dataclass(frozen=True)
class PlacementResult:
    coordinate: np.ndarray

    distance_1: float
    distance_2: float

    residual_1: float
    residual_2: float

    objective: float

    sulfur_distance: float

    exact_geometry_possible: bool

    method: str

    steric_score: float

def normalize_protein_name(name: str) -> str:
    """
    Convert, for example:

        OR2M3 Human

    to:

        OR2M3
    """

    return re.sub(
        r"\s+Human$",
        "",
        str(name).strip(),
        flags=re.IGNORECASE,
    )


def find_protein_column(
    df: pd.DataFrame,
    protein: str,
) -> str:
    """
    Match a command-line protein name such as:

        OR2M3

    against an Excel column such as:

        OR2M3 Human
    """

    protein_norm = protein.strip().lower()

    candidates: List[str] = []

    for col in df.columns:
        col_str = str(col).strip()
        col_norm = col_str.lower()

        if (
            col_norm == protein_norm
            or col_norm.startswith(protein_norm + " ")
        ):
            return col_str

        if col_norm.startswith(protein_norm):
            candidates.append(col_str)

    if len(candidates) == 1:
        return candidates[0]

    if len(candidates) > 1:
        raise ValueError(
            f"Protein {protein!r} matched multiple columns: "
            f"{candidates}"
        )

    raise ValueError(
        f"Protein {protein!r} not found in columns: "
        f"{list(df.columns)}"
    )


def is_segment_label(value: object) -> bool:
    """
    Determine whether a value in the GPCRdb(A) column is a segment label
    such as:

        TM1
        TM2
        ECL1
        ICL2
        H8

    rather than a generic position such as:

        5x42
    """

    text = str(value).strip()

    if not text:
        return False

    if text.lower() == "nan":
        return False

    if "x" in text:
        return False

    return bool(
        re.fullmatch(
            r"(TM\d+|ICL\d+|ECL\d+|N-term|C-term|H8|EL\d+|IL\d+)",
            text,
            flags=re.IGNORECASE,
        )
    )


def parse_aa_and_resid(
    cell: object,
) -> Tuple[Optional[str], Optional[int]]:
    """
    Parse residue labels from Excel.

    Examples:

        C202     -> ("C", 202)
        C 202    -> ("C", 202)
        202      -> (None, 202)
        -        -> (None, None)
    """

    if cell is None:
        return None, None

    if isinstance(cell, float) and pd.isna(cell):
        return None, None

    text = str(cell).strip()

    if text in {
        "",
        "-",
        "nan",
        "NaN",
    }:
        return None, None

    match = re.fullmatch(
        r"\s*([A-Za-z]+)?\s*(\d+)\s*",
        text,
    )

    if match:
        aa_raw = match.group(1)

        if aa_raw:
            aa = aa_raw[0].upper()
        else:
            aa = None

        residue_number = int(match.group(2))

        return aa, residue_number

    number_match = re.search(
        r"(\d+)",
        text,
    )

    if not number_match:
        return None, None

    aa_match = re.search(
        r"([A-Za-z]+)",
        text,
    )

    if aa_match:
        aa = aa_match.group(1)[0].upper()
    else:
        aa = None

    residue_number = int(
        number_match.group(1)
    )

    return aa, residue_number


def load_protein_mapping(
    excel_path: Path,
    sheet_name: str,
    protein: str,
) -> Tuple[str, Dict[str, ResidueInfo]]:
    """
    Read the Excel table and build a generic-position mapping for one protein.

    Returns:

        protein_column,
        mapping

    where:

        mapping["5x42"]

    returns a ResidueInfo object.
    """

    df = pd.read_excel(
        excel_path,
        sheet_name=sheet_name,
    )

    if "GPCRdb(A)" not in df.columns:
        raise ValueError(
            f"Column 'GPCRdb(A)' not found in {excel_path}. "
            f"Available columns: {list(df.columns)}"
        )

    protein_column = find_protein_column(
        df,
        protein,
    )

    mapping: Dict[str, ResidueInfo] = {}

    current_segment = "UNKNOWN"

    for _, row in df.iterrows():

        generic_or_segment = row.get(
            "GPCRdb(A)",
            None,
        )

        if is_segment_label(generic_or_segment):

            current_segment = str(
                generic_or_segment
            ).strip()

            continue

        generic = str(
            generic_or_segment
        ).strip()

        if not generic:
            continue

        if generic.lower() == "nan":
            continue

        if "x" not in generic:
            continue

        aa, residue_number = parse_aa_and_resid(
            row.get(
                protein_column,
                None,
            )
        )

        if residue_number is None:
            continue

        if aa is None:
            aa = "X"

        mapping[generic] = ResidueInfo(
            generic_position=generic,
            residue_label=f"{aa}{residue_number}",
            residue_name_1=aa,
            residue_number=residue_number,
            segment=current_segment,
        )

    return protein_column, mapping

def infer_element(
    atom_name: str,
    element_field: str,
) -> str:
    """
    Determine the chemical element.

    Prefer the official PDB element column.

    If that column is empty, infer from the atom name.
    """

    element = element_field.strip().upper()

    if element:
        return element

    cleaned = re.sub(
        r"[^A-Za-z]",
        "",
        atom_name,
    ).upper()

    if not cleaned:
        return ""

    # For the protein atoms considered here, the first alphabetic
    # character is adequate for identifying H vs heavy atoms
    # and sulfur atoms.
    return cleaned[0]


def parse_pdb_atoms(
    pdb_path: Path,
    model_number: int = 1,
) -> List[AtomRecord]:
    """
    Parse ATOM/HETATM records from one PDB model.
    """

    atoms: List[AtomRecord] = []

    current_model = 1

    saw_model_record = False

    selected_model_finished = False

    with pdb_path.open(
        "r",
        encoding="utf-8",
        errors="replace",
    ) as handle:

        for line in handle:

            record = line[0:6].strip()

            if record == "MODEL":

                saw_model_record = True

                try:
                    current_model = int(
                        line[10:14].strip()
                    )

                except ValueError:
                    current_model += 1

                continue

            if record == "ENDMDL":

                if (
                    saw_model_record
                    and current_model == model_number
                ):
                    selected_model_finished = True

                if selected_model_finished:
                    break

                continue

            if record not in {
                "ATOM",
                "HETATM",
            }:
                continue

            if (
                saw_model_record
                and current_model != model_number
            ):
                continue

            try:

                atom = AtomRecord(
                    record_name=record,

                    serial=int(
                        line[6:11]
                    ),

                    atom_name=line[
                        12:16
                    ].strip(),

                    altloc=line[
                        16:17
                    ].strip(),

                    residue_name=line[
                        17:20
                    ].strip(),

                    chain_id=line[
                        21:22
                    ].strip(),

                    residue_number=int(
                        line[22:26]
                    ),

                    insertion_code=line[
                        26:27
                    ].strip(),

                    coord=np.array(
                        [
                            float(line[30:38]),
                            float(line[38:46]),
                            float(line[46:54]),
                        ],
                        dtype=float,
                    ),

                    occupancy=float(
                        line[54:60].strip()
                        or 0.0
                    ),

                    b_factor=float(
                        line[60:66].strip()
                        or 0.0
                    ),

                    element=infer_element(
                        line[12:16],

                        line[76:78]
                        if len(line) >= 78
                        else "",
                    ),
                )

            except (
                ValueError,
                IndexError,
            ) as exc:

                raise ValueError(
                    "Could not parse PDB atom line:\n"
                    f"{line.rstrip()}"
                ) from exc

            atoms.append(atom)

    if not atoms:
        raise ValueError(
            f"No atoms were parsed from model "
            f"{model_number} of {pdb_path}"
        )

    return atoms


# =============================================================================
# CYSTEINE SULFUR SELECTION
# =============================================================================


def select_unique_cysteine_sulfur_atom(
    atoms: Sequence[AtomRecord],
    chain_id: str,
    residue_number: int,
    atom_name_regex: str = CYSTEINE_SULFUR_ATOM_REGEX,
) -> AtomRecord:
    """
    Select the unique sulfur atom in the mapped cysteine residue.

    The function does NOT assume that the sulfur atom is named "SG".

    Requirements:

        1. Correct chain ID.
        2. Correct residue number.
        3. Chemical element is sulfur.
        4. Atom name matches CYSTEINE_SULFUR_ATOM_REGEX.
        5. altloc is either blank or A.

    Because cysteine contains one sulfur atom, multiple matching sulfur
    atoms are treated as an ambiguity.
    """

    try:

        name_pattern = re.compile(
            atom_name_regex,
            flags=re.IGNORECASE,
        )

    except re.error as exc:

        raise ValueError(
            f"Invalid CYSTEINE_SULFUR_ATOM_REGEX "
            f"{atom_name_regex!r}: {exc}"
        ) from exc

    matches = [
        atom
        for atom in atoms

        if (
            atom.chain_id == chain_id

            and atom.residue_number == residue_number

            and atom.element.upper() == "S"

            and name_pattern.fullmatch(
                atom.atom_name.strip()
            ) is not None

            and atom.altloc in {
                "",
                "A",
            }
        )
    ]

    if not matches:

        atoms_in_residue = [
            atom
            for atom in atoms

            if (
                atom.chain_id == chain_id

                and atom.residue_number == residue_number

                and atom.altloc in {
                    "",
                    "A",
                }
            )
        ]

        available = ", ".join(
            (
                f"{atom.atom_name}"
                f"(element={atom.element or '?'})"
            )
            for atom in atoms_in_residue
        )

        if not available:
            available = "none"

        raise ValueError(
            f"No sulfur atom matching regex "
            f"{atom_name_regex!r} was found for "
            f"chain {chain_id!r}, "
            f"residue {residue_number}. "
            f"Available atoms: {available}"
        )

    # Prefer blank altloc over altloc A.
    matches.sort(
        key=lambda atom: (
            atom.altloc != "",
            atom.serial,
        )
    )

    best = matches[0]

    equivalent = [
        atom
        for atom in matches
        if atom.altloc == best.altloc
    ]

    if len(equivalent) > 1:

        identifiers = "\n  ".join(
            atom.identifier
            for atom in equivalent
        )

        raise ValueError(
            "Multiple sulfur atoms matched in a residue "
            "expected to contain one:\n  "
            + identifiers
        )

    return best


# =============================================================================
# GEOMETRY
# =============================================================================


def make_perpendicular_basis(
    axis: np.ndarray,
) -> Tuple[np.ndarray, np.ndarray]:
    """
    Construct two orthonormal vectors perpendicular to an axis.

    These vectors define the plane of the sphere-intersection circle.
    """

    unit_axis = axis / np.linalg.norm(axis)

    candidates = np.eye(3)

    seed = candidates[
        np.argmin(
            np.abs(
                candidates @ unit_axis
            )
        )
    ]

    u = np.cross(
        unit_axis,
        seed,
    )

    u /= np.linalg.norm(u)

    v = np.cross(
        unit_axis,
        u,
    )

    v /= np.linalg.norm(v)

    return u, v


# =============================================================================
# STERIC / CROWDING SCORE
# =============================================================================


def steric_score(
    candidate: np.ndarray,
    heavy_coords: np.ndarray,
) -> float:
    """
    Calculate a geometric crowding score for one candidate Cu position.

    Lower is better.

    The score has two parts:

    1. Very large penalty if a non-coordinating heavy atom is closer than:

           MIN_NONBOND_CLEARANCE_A

    2. Smooth crowding penalty from nearby heavy atoms:

           exp(
               -0.5 *
               (distance / CROWDING_LENGTH_SCALE_A)^2
           )

    Atoms farther than CROWDING_CUTOFF_A are ignored.

    IMPORTANT:
    This is a geometric heuristic, not a molecular-mechanics or QM energy.
    """

    if heavy_coords.size == 0:
        return 0.0

    distances = np.linalg.norm(
        heavy_coords - candidate,
        axis=1,
    )

    local = distances[
        distances <= CROWDING_CUTOFF_A
    ]

    if local.size == 0:
        return 0.0

    clash = np.maximum(
        0.0,
        MIN_NONBOND_CLEARANCE_A - local,
    )

    clash_penalty = (
        1.0e6
        * float(
            np.sum(
                clash * clash
            )
        )
    )

    crowding_penalty = float(
        np.sum(
            np.exp(
                -0.5
                * (
                    local
                    / CROWDING_LENGTH_SCALE_A
                ) ** 2
            )
        )
    )

    return (
        clash_penalty
        + crowding_penalty
    )


# =============================================================================
# SPHERE-INTERSECTION CIRCLE
# =============================================================================


def choose_open_point_on_intersection_circle(
    sulfur_1: np.ndarray,
    sulfur_2: np.ndarray,
    radius_1: float,
    radius_2: float,
    heavy_coords: np.ndarray,
) -> Tuple[np.ndarray, float, str]:
    """
    Find a Cu coordinate when the two target distance spheres intersect.

    Every point on the intersection circle satisfies:

        |Cu - S1| = radius_1

    and:

        |Cu - S2| = radius_2

    exactly.

    Because those two constraints do not uniquely specify a coordinate,
    points around the circle are evaluated and the least-crowded position
    is selected.
    """

    displacement = (
        sulfur_2
        - sulfur_1
    )

    sulfur_distance = float(
        np.linalg.norm(
            displacement
        )
    )

    axis = (
        displacement
        / sulfur_distance
    )

    axial_distance = (
        sulfur_distance * sulfur_distance
        + radius_1 * radius_1
        - radius_2 * radius_2
    ) / (
        2.0 * sulfur_distance
    )

    circle_center = (
        sulfur_1
        + axial_distance * axis
    )

    circle_radius_sq = (
        radius_1 * radius_1
        - axial_distance * axial_distance
    )

    circle_radius = math.sqrt(
        max(
            0.0,
            circle_radius_sq,
        )
    )

    # If the spheres are tangent, the circle radius is effectively zero.
    if circle_radius < 1.0e-10:

        return (
            circle_center,
            steric_score(
                circle_center,
                heavy_coords,
            ),
            "tangent spheres",
        )

    u, v = make_perpendicular_basis(
        axis
    )

    best_coord: Optional[np.ndarray] = None

    best_score = math.inf

    best_theta = 0.0

    # ---------------------------------------------------------------------
    # Initial deterministic circle sampling
    # ---------------------------------------------------------------------

    for index in range(
        N_CIRCLE_SAMPLES
    ):

        theta = (
            2.0
            * math.pi
            * index
            / N_CIRCLE_SAMPLES
        )

        candidate = (
            circle_center
            + circle_radius
            * (
                math.cos(theta) * u
                + math.sin(theta) * v
            )
        )

        score = steric_score(
            candidate,
            heavy_coords,
        )

        if score < best_score:

            best_score = score
            best_coord = candidate
            best_theta = theta

    assert best_coord is not None

    # ---------------------------------------------------------------------
    # Continuous local angular refinement around the best sampled point
    # ---------------------------------------------------------------------

    step = (
        2.0
        * math.pi
        / N_CIRCLE_SAMPLES
    )

    left = (
        best_theta
        - step
    )

    right = (
        best_theta
        + step
    )

    golden = (
        math.sqrt(5.0)
        - 1.0
    ) / 2.0

    x1 = (
        right
        - golden
        * (
            right
            - left
        )
    )

    x2 = (
        left
        + golden
        * (
            right
            - left
        )
    )

    def candidate_at(
        theta: float,
    ) -> np.ndarray:

        return (
            circle_center
            + circle_radius
            * (
                math.cos(theta) * u
                + math.sin(theta) * v
            )
        )

    f1 = steric_score(
        candidate_at(x1),
        heavy_coords,
    )

    f2 = steric_score(
        candidate_at(x2),
        heavy_coords,
    )

    for _ in range(60):

        if f1 <= f2:

            right = x2

            x2 = x1
            f2 = f1

            x1 = (
                right
                - golden
                * (
                    right
                    - left
                )
            )

            f1 = steric_score(
                candidate_at(x1),
                heavy_coords,
            )

        else:

            left = x1

            x1 = x2
            f1 = f2

            x2 = (
                left
                + golden
                * (
                    right
                    - left
                )
            )

            f2 = steric_score(
                candidate_at(x2),
                heavy_coords,
            )

    theta = 0.5 * (
        left
        + right
    )

    refined = candidate_at(
        theta
    )

    refined_score = steric_score(
        refined,
        heavy_coords,
    )

    return (
        refined,
        refined_score,
        (
            "sphere-intersection circle "
            "+ open-space tie-breaker"
        ),
    )


# =============================================================================
# LEAST-SQUARES CASE
# =============================================================================


def least_squares_axis_point(
    sulfur_1: np.ndarray,
    sulfur_2: np.ndarray,
    radius_1: float,
    radius_2: float,
) -> np.ndarray:
    """
    Return the global fixed-sulfur distance least-squares solution when
    the two target-distance spheres do not intersect.

    The minimized objective is:

        sqrt(
            (d1 - radius_1)^2
            +
            (d2 - radius_2)^2
        )

    where:

        d1 = |Cu - S1|
        d2 = |Cu - S2|
    """

    displacement = (
        sulfur_2
        - sulfur_1
    )

    sulfur_distance = float(
        np.linalg.norm(
            displacement
        )
    )

    if sulfur_distance < 1.0e-12:

        # Coincident sulfur atoms:
        # every direction is equivalent.
        #
        # Use +x and the average target radius.

        return (
            sulfur_1
            + np.array(
                [
                    (
                        radius_1
                        + radius_2
                    )
                    / 2.0,
                    0.0,
                    0.0,
                ]
            )
        )

    axis = (
        displacement
        / sulfur_distance
    )

    # Evaluate the stationary point in each of the three 1D regions along
    # the sulfur-sulfur axis, plus the two region boundaries.
    #
    # This handles:
    #
    #   - non-overlapping spheres
    #   - one sphere lying completely inside the other
    #

    candidate_t = {
        0.0,

        sulfur_distance,

        min(
            sulfur_distance,
            max(
                0.0,
                (
                    sulfur_distance
                    + radius_1
                    - radius_2
                )
                / 2.0,
            ),
        ),

        (
            sulfur_distance
            + radius_1
            + radius_2
        )
        / 2.0,

        (
            sulfur_distance
            - radius_1
            - radius_2
        )
        / 2.0,
    }

    best_point: Optional[np.ndarray] = None

    best_objective = math.inf

    for t in candidate_t:

        point = (
            sulfur_1
            + t * axis
        )

        d1 = float(
            np.linalg.norm(
                point
                - sulfur_1
            )
        )

        d2 = float(
            np.linalg.norm(
                point
                - sulfur_2
            )
        )

        objective = math.hypot(
            d1 - radius_1,
            d2 - radius_2,
        )

        if objective < best_objective:

            best_objective = objective

            best_point = point

    assert best_point is not None

    return best_point


# =============================================================================
# MAIN Cu PLACEMENT
# =============================================================================


def place_copper(
    sulfur_1: np.ndarray,
    sulfur_2: np.ndarray,
    radius_1: float,
    radius_2: float,
    heavy_coords: np.ndarray,
) -> PlacementResult:
    """
    Compute the Cu(I) coordinate.
    """

    sulfur_distance = float(
        np.linalg.norm(
            sulfur_2
            - sulfur_1
        )
    )

    lower_limit = abs(
        radius_1
        - radius_2
    )

    upper_limit = (
        radius_1
        + radius_2
    )

    tolerance = 1.0e-10

    exact_possible = (
        lower_limit - tolerance
        <= sulfur_distance
        <= upper_limit + tolerance
    )

    if (
        exact_possible
        and sulfur_distance > tolerance
    ):

        (
            coordinate,
            score,
            method,
        ) = choose_open_point_on_intersection_circle(
            sulfur_1,
            sulfur_2,
            radius_1,
            radius_2,
            heavy_coords,
        )

    else:

        coordinate = least_squares_axis_point(
            sulfur_1,
            sulfur_2,
            radius_1,
            radius_2,
        )

        score = steric_score(
            coordinate,
            heavy_coords,
        )

        method = (
            "fixed-sulfur global distance "
            "least-squares point on sulfur-sulfur axis"
        )

    distance_1 = float(
        np.linalg.norm(
            coordinate
            - sulfur_1
        )
    )

    distance_2 = float(
        np.linalg.norm(
            coordinate
            - sulfur_2
        )
    )

    residual_1 = (
        distance_1
        - radius_1
    )

    residual_2 = (
        distance_2
        - radius_2
    )

    objective = math.hypot(
        residual_1,
        residual_2,
    )

    return PlacementResult(
        coordinate=coordinate,

        distance_1=distance_1,
        distance_2=distance_2,

        residual_1=residual_1,
        residual_2=residual_2,

        objective=objective,

        sulfur_distance=sulfur_distance,

        exact_geometry_possible=exact_possible,

        method=method,

        steric_score=score,
    )


# =============================================================================
# PDB OUTPUT FORMATTING
# =============================================================================


def pdb_coordinate_tail(
    coordinate: np.ndarray,
    occupancy: float,
    b_factor: float,
    element: str,
) -> str:
    """
    Return only PDB columns 31-78:

        x
        y
        z
        occupancy
        B-factor
        element

    Example:

          -4.073  10.630 -16.940  1.00  0.00          CU
    """

    x, y, z = coordinate

    return (
        f"{x:8.3f}"
        f"{y:8.3f}"
        f"{z:8.3f}"
        f"{occupancy:6.2f}"
        f"{b_factor:6.2f}"
        f"          "
        f"{element:>2s}"
    )


# =============================================================================
# NEAREST-ATOM DIAGNOSTIC
# =============================================================================


def nearest_noncoordinating_atom(
    coordinate: np.ndarray,
    atoms: Iterable[AtomRecord],
    excluded_serials: set[int],
) -> Tuple[Optional[AtomRecord], float]:
    """
    Find the nearest non-hydrogen atom to the Cu coordinate,
    excluding the two coordinating cysteine sulfur atoms.
    """

    best_atom: Optional[AtomRecord] = None

    best_distance = math.inf

    for atom in atoms:

        if atom.serial in excluded_serials:
            continue

        if atom.element in {
            "H",
            "D",
        }:
            continue

        distance = float(
            np.linalg.norm(
                atom.coord
                - coordinate
            )
        )

        if distance < best_distance:

            best_distance = distance
            best_atom = atom

    return (
        best_atom,
        best_distance,
    )


# =============================================================================
# OUTPUT REPORT
# =============================================================================


def write_output(
    output_path: Path,
    pdb_path: Path,
    protein_column: str,
    residue_1: ResidueInfo,
    residue_2: ResidueInfo,
    sulfur_atom_1: AtomRecord,
    sulfur_atom_2: AtomRecord,
    result: PlacementResult,
    nearest_atom: Optional[AtomRecord],
    nearest_distance: float,
) -> None:
    """
    Write Cu coordinate plus diagnostic information to a text file.
    """

    radius_1, radius_2 = (
        REFERENCE_DISTANCES_A
    )

    lower_limit = abs(
        radius_1
        - radius_2
    )

    upper_limit = (
        radius_1
        + radius_2
    )

    tail = pdb_coordinate_tail(
        result.coordinate,
        occupancy=CU_OCCUPANCY,
        b_factor=CU_B_FACTOR,
        element=CU_ELEMENT,
    )

    output_path.parent.mkdir(
        parents=True,
        exist_ok=True,
    )

    with output_path.open(
        "w",
        encoding="utf-8",
    ) as handle:

        # ---------------------------------------------------------------
        # Put the requested PDB coordinate portion FIRST.
        # ---------------------------------------------------------------

        handle.write(
            tail
            + "\n"
        )

        handle.write(
            "# PDB coordinate tail above is columns 31-78 only.\n"
        )

        handle.write(
            "# Prepend/edit HETATM, serial, atom name, residue name, "
            "chain, and residue ID as needed.\n"
        )

        handle.write(
            f"# Protein mapping column: "
            f"{protein_column}\n"
        )

        handle.write(
            f"# Input PDB: "
            f"{pdb_path}\n"
        )

        handle.write(
            f"# Chain/model: "
            f"{CHAIN_ID!r} / "
            f"{MODEL_NUMBER}\n"
        )

        handle.write(
            "# Cysteine sulfur atom-name regex: "
            f"{CYSTEINE_SULFUR_ATOM_REGEX!r}\n"
        )

        # ---------------------------------------------------------------
        # Cysteine 1
        # ---------------------------------------------------------------

        handle.write(
            f"# Cys 1: "
            f"{residue_1.generic_position} "
            f"-> "
            f"{residue_1.residue_label} "
            f"({residue_1.segment}); "
            f"{sulfur_atom_1.identifier}; "
            f"sulfur xyz = "
            f"{sulfur_atom_1.coord[0]:.6f} "
            f"{sulfur_atom_1.coord[1]:.6f} "
            f"{sulfur_atom_1.coord[2]:.6f}\n"
        )

        # ---------------------------------------------------------------
        # Cysteine 2
        # ---------------------------------------------------------------

        handle.write(
            f"# Cys 2: "
            f"{residue_2.generic_position} "
            f"-> "
            f"{residue_2.residue_label} "
            f"({residue_2.segment}); "
            f"{sulfur_atom_2.identifier}; "
            f"sulfur xyz = "
            f"{sulfur_atom_2.coord[0]:.6f} "
            f"{sulfur_atom_2.coord[1]:.6f} "
            f"{sulfur_atom_2.coord[2]:.6f}\n"
        )

        # ---------------------------------------------------------------
        # Reference distances
        # ---------------------------------------------------------------

        handle.write(
            "# Reference Cu-S distances (angstrom): "
            f"{radius_1:.6f} "
            f"{radius_2:.6f}\n"
        )

        # ---------------------------------------------------------------
        # Actual calculated distances
        # ---------------------------------------------------------------

        handle.write(
            "# Computed Cu-S distances (angstrom): "
            f"{result.distance_1:.6f} "
            f"{result.distance_2:.6f}\n"
        )

        # ---------------------------------------------------------------
        # Residuals
        # ---------------------------------------------------------------

        handle.write(
            "# Distance residuals computed-reference (angstrom): "
            f"{result.residual_1:.6f} "
            f"{result.residual_2:.6f}\n"
        )

        # ---------------------------------------------------------------
        # Optimization value
        # ---------------------------------------------------------------

        handle.write(
            "# Optimization value "
            "sqrt((d1-r1)^2 + (d2-r2)^2) "
            "(angstrom): "
            f"{result.objective:.6f}\n"
        )

        # ---------------------------------------------------------------
        # Sulfur-sulfur distance
        # ---------------------------------------------------------------

        handle.write(
            "# Cys sulfur-sulfur distance (angstrom): "
            f"{result.sulfur_distance:.6f}\n"
        )

        # ---------------------------------------------------------------
        # Exact geometry
        # ---------------------------------------------------------------

        handle.write(
            "# Exact two-distance geometry possible with fixed "
            "sulfur atoms: "
            f"{'YES' if result.exact_geometry_possible else 'NO'}\n"
        )

        handle.write(
            "# Exact sphere-intersection condition: "
            f"{lower_limit:.6f} "
            "<= sulfur-sulfur <= "
            f"{upper_limit:.6f} "
            "angstrom\n"
        )

        # ---------------------------------------------------------------
        # Placement method
        # ---------------------------------------------------------------

        handle.write(
            "# Placement method: "
            f"{result.method}\n"
        )

        # ---------------------------------------------------------------
        # Steric diagnostic
        # ---------------------------------------------------------------

        handle.write(
            "# Steric/crowding diagnostic score "
            "(not part of the reported distance objective): "
            f"{result.steric_score:.6f}\n"
        )

        # ---------------------------------------------------------------
        # Nearest atom
        # ---------------------------------------------------------------

        if nearest_atom is not None:

            handle.write(
                "# Nearest non-coordinating heavy atom: "
                f"{nearest_atom.identifier}; "
                f"distance = "
                f"{nearest_distance:.6f} "
                "angstrom\n"
            )

            if (
                nearest_distance
                < MIN_NONBOND_CLEARANCE_A
            ):

                handle.write(
                    "# WARNING: nearest-heavy-atom distance "
                    "is below the configured "
                    f"{MIN_NONBOND_CLEARANCE_A:.2f} "
                    "angstrom clearance.\n"
                )

        # ---------------------------------------------------------------
        # Warning when exact target geometry cannot be reached
        # ---------------------------------------------------------------

        if not result.exact_geometry_possible:

            if (
                result.sulfur_distance
                > upper_limit
            ):

                gap = (
                    result.sulfur_distance
                    - upper_limit
                )

                handle.write(
                    "# WARNING: sulfur atoms are "
                    f"{gap:.6f} "
                    "angstrom farther apart than the maximum "
                    "distance compatible with both reference "
                    "Cu-S distances.\n"
                )

            else:

                gap = (
                    lower_limit
                    - result.sulfur_distance
                )

                handle.write(
                    "# WARNING: sulfur atoms are "
                    f"{gap:.6f} "
                    "angstrom too close for the two unequal "
                    "reference Cu-S distances.\n"
                )

            handle.write(
                "# The coordinate is therefore a mathematical "
                "least-squares compromise, not a chemically "
                "validated Cu(I) site. "
                "Side-chain/ligand refinement is required.\n"
            )


# =============================================================================
# COMMAND-LINE ARGUMENTS
# =============================================================================


def parse_command_line() -> argparse.Namespace:
    """
    Read:

        protein
        pdb_file

    from the terminal.

    Examples:

        python3 script.py OR2M3 OR2M3_Prep.pdb

    or:

        python3 script.py \
            OR2M3 \
            OR2M3_Prep.pdb \
            -o OR2M3_Cu.txt
    """

    parser = argparse.ArgumentParser(
        description=(
            "Place a Cu(I) coordinate using the two mapped "
            "cysteine sulfur atoms. "
            "PROTEIN and PDB_FILE are supplied as positional "
            "command-line arguments."
        )
    )

    parser.add_argument(
        "protein",
        help=(
            "Protein name used to select the Excel mapping column, "
            'e.g. "OR2M3".'
        ),
    )

    parser.add_argument(
        "pdb_file",
        type=Path,
        help=(
            "Input PDB file, "
            'e.g. "OR2M3_Prep.pdb".'
        ),
    )

    parser.add_argument(
        "-o",
        "--output",
        type=Path,
        default=None,
        help=(
            "Optional output text-file path. "
            "By default, the file is written next to the PDB as "
            "<PROTEIN>_site1_Cu_coordinate.txt."
        ),
    )

    return parser.parse_args()


def sanitize_filename(
    text: str,
) -> str:
    """
    Make a protein name safe for use in an output filename.
    """

    return re.sub(
        r"[^A-Za-z0-9_.-]+",
        "_",
        text.strip(),
    )


# =============================================================================
# MAIN
# =============================================================================


def main() -> None:

    # ---------------------------------------------------------------------
    # Read terminal arguments
    # ---------------------------------------------------------------------

    args = parse_command_line()

    protein = args.protein.strip()

    pdb_path = (
        args.pdb_file
        .expanduser()
    )

    # ---------------------------------------------------------------------
    # Decide output filename
    # ---------------------------------------------------------------------

    if args.output is None:

        output_path = (
            pdb_path.parent
            / (
                f"{sanitize_filename(protein)}"
                "_site1_Cu_coordinate.txt"
            )
        )

    else:

        output_path = (
            args.output
            .expanduser()
        )

    # ---------------------------------------------------------------------
    # Check input PDB
    # ---------------------------------------------------------------------

    if not pdb_path.exists():

        raise FileNotFoundError(
            f"PDB file not found: "
            f"{pdb_path}"
        )

    # ---------------------------------------------------------------------
    # Validate selection mode
    # ---------------------------------------------------------------------

    if SELECTION_MODE not in {
        "generic",
        "residue_numbers",
    }:

        raise ValueError(
            "SELECTION_MODE must be "
            "'generic' or 'residue_numbers'."
        )

    # ---------------------------------------------------------------------
    # Determine the two cysteine residues
    # ---------------------------------------------------------------------

    if SELECTION_MODE == "generic":

        if not EXCEL_PATH.exists():

            raise FileNotFoundError(
                f"Excel mapping file not found: "
                f"{EXCEL_PATH}"
            )

        (
            protein_column,
            mapping,
        ) = load_protein_mapping(
            EXCEL_PATH,
            SHEET_NAME,
            protein,
        )

        missing = [
            position
            for position
            in GENERIC_CYS_POSITIONS

            if position not in mapping
        ]

        if missing:

            raise KeyError(
                f"Missing generic positions "
                f"for {protein_column}: "
                f"{missing}"
            )

        residue_1 = mapping[
            GENERIC_CYS_POSITIONS[0]
        ]

        residue_2 = mapping[
            GENERIC_CYS_POSITIONS[1]
        ]

    else:

        protein_column = (
            f"{protein} "
            "(direct residue-number mode)"
        )

        residue_1 = ResidueInfo(
            generic_position="DIRECT_1",
            residue_label=(
                f"C{DIRECT_CYS_RESIDUES[0]}"
            ),
            residue_name_1="C",
            residue_number=(
                DIRECT_CYS_RESIDUES[0]
            ),
            segment="UNKNOWN",
        )

        residue_2 = ResidueInfo(
            generic_position="DIRECT_2",
            residue_label=(
                f"C{DIRECT_CYS_RESIDUES[1]}"
            ),
            residue_name_1="C",
            residue_number=(
                DIRECT_CYS_RESIDUES[1]
            ),
            segment="UNKNOWN",
        )

    # ---------------------------------------------------------------------
    # Validate generic mapping really maps to cysteine
    # ---------------------------------------------------------------------

    if REQUIRE_CYSTEINE:

        for residue in (
            residue_1,
            residue_2,
        ):

            if (
                residue.residue_name_1
                != "C"
            ):

                raise ValueError(
                    f"{residue.generic_position} "
                    f"maps to "
                    f"{residue.residue_label}, "
                    "not cysteine. "
                    "Check the numbering convention "
                    "and GENERIC_CYS_POSITIONS."
                )

    # ---------------------------------------------------------------------
    # Read PDB
    # ---------------------------------------------------------------------

    atoms = parse_pdb_atoms(
        pdb_path,
        model_number=MODEL_NUMBER,
    )

    # ---------------------------------------------------------------------
    # Find sulfur atom in Cys 1
    # ---------------------------------------------------------------------

    sulfur_atom_1 = (
        select_unique_cysteine_sulfur_atom(
            atoms,
            CHAIN_ID,
            residue_1.residue_number,
        )
    )

    # ---------------------------------------------------------------------
    # Find sulfur atom in Cys 2
    # ---------------------------------------------------------------------

    sulfur_atom_2 = (
        select_unique_cysteine_sulfur_atom(
            atoms,
            CHAIN_ID,
            residue_2.residue_number,
        )
    )

    # ---------------------------------------------------------------------
    # Validate PDB residue names are cysteine-like
    # ---------------------------------------------------------------------

    if REQUIRE_CYSTEINE:

        try:

            cysteine_resname_pattern = re.compile(
                CYSTEINE_RESIDUE_NAME_REGEX,
                flags=re.IGNORECASE,
            )

        except re.error as exc:

            raise ValueError(
                "Invalid CYSTEINE_RESIDUE_NAME_REGEX "
                f"{CYSTEINE_RESIDUE_NAME_REGEX!r}: "
                f"{exc}"
            ) from exc

        for (
            residue,
            sulfur_atom,
        ) in (
            (
                residue_1,
                sulfur_atom_1,
            ),
            (
                residue_2,
                sulfur_atom_2,
            ),
        ):

            if (
                cysteine_resname_pattern.fullmatch(
                    sulfur_atom.residue_name
                )
                is None
            ):

                raise ValueError(
                    f"Mapped "
                    f"{residue.generic_position}/"
                    f"{residue.residue_label}, "
                    "but the PDB sulfur is in residue "
                    f"{sulfur_atom.residue_name!r}, "
                    "which does not match "
                    "CYSTEINE_RESIDUE_NAME_REGEX="
                    f"{CYSTEINE_RESIDUE_NAME_REGEX!r}. "
                    "Check residue numbering or adjust "
                    "the residue-name regex for the force field."
                )

    # ---------------------------------------------------------------------
    # Construct list of non-coordinating heavy atoms
    # ---------------------------------------------------------------------

    excluded_serials = {
        sulfur_atom_1.serial,
        sulfur_atom_2.serial,
    }

    heavy_atoms = [
        atom
        for atom in atoms

        if (
            atom.serial
            not in excluded_serials

            and atom.element
            not in {
                "H",
                "D",
            }
        )
    ]

    heavy_coords = np.array(
        [
            atom.coord
            for atom in heavy_atoms
        ],
        dtype=float,
    )

    # ---------------------------------------------------------------------
    # Calculate Cu coordinate
    # ---------------------------------------------------------------------

    result = place_copper(
        sulfur_atom_1.coord,
        sulfur_atom_2.coord,

        REFERENCE_DISTANCES_A[0],
        REFERENCE_DISTANCES_A[1],

        heavy_coords,
    )

    # ---------------------------------------------------------------------
    # Find nearest non-coordinating protein heavy atom
    # ---------------------------------------------------------------------

    (
        nearest_atom,
        nearest_distance,
    ) = nearest_noncoordinating_atom(
        result.coordinate,
        atoms,
        excluded_serials,
    )

    # ---------------------------------------------------------------------
    # Write output
    # ---------------------------------------------------------------------

    write_output(
        output_path=output_path,

        pdb_path=pdb_path,

        protein_column=protein_column,

        residue_1=residue_1,
        residue_2=residue_2,

        sulfur_atom_1=sulfur_atom_1,
        sulfur_atom_2=sulfur_atom_2,

        result=result,

        nearest_atom=nearest_atom,
        nearest_distance=nearest_distance,
    )

    # ---------------------------------------------------------------------
    # Terminal summary
    # ---------------------------------------------------------------------

    print(
        f"Protein: "
        f"{protein}"
    )

    print(
        f"Input PDB: "
        f"{pdb_path}"
    )

    print(
        f"Wrote "
        f"{output_path}"
    )

    print(
        "Cu coordinate: "
        f"{result.coordinate[0]:.6f} "
        f"{result.coordinate[1]:.6f} "
        f"{result.coordinate[2]:.6f}"
    )

    print(
        "Computed distances: "
        f"{result.distance_1:.6f} "
        f"{result.distance_2:.6f} "
        "angstrom"
    )

    print(
        "Optimization value: "
        f"{result.objective:.6f} "
        "angstrom"
    )

    print(
        "Exact fixed-sulfur geometry possible: "
        f"{result.exact_geometry_possible}"
    )


if __name__ == "__main__":
    main()
